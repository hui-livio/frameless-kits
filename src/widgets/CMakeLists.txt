qt_add_library(fk_widgets ${FRAMELESSKIT_LIBRARY_TYPE})
framelesskit_configure_target(fk_widgets Widgets)
set_target_properties(fk_widgets PROPERTIES OUTPUT_NAME framelesswidget)
framelesskit_embed_core(fk_widgets)
target_sources(fk_widgets PRIVATE WidgetAgent.cpp "${PROJECT_SOURCE_DIR}/include/FramelessKit/WidgetAgent.h")
if(APPLE)
    target_sources(fk_widgets PRIVATE CocoaWidgetSafeArea_p.h)
endif()
target_link_libraries(fk_widgets PUBLIC Qt6::Widgets)
if(FRAMELESSKIT_INSTALL)
    install(FILES "${PROJECT_SOURCE_DIR}/include/FramelessKit/WidgetAgent.h" DESTINATION "${CMAKE_INSTALL_INCLUDEDIR}/FramelessKit")
endif()
