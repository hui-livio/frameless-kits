cmake_minimum_required(VERSION 3.24)
project(WindowHelerInstalledConsumer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CONSUMER_FRONTEND "Widgets" CACHE STRING "Widgets, Quick, Mixed, or MixedQuickFirst")
if(CONSUMER_FRONTEND STREQUAL "MixedQuickFirst")
    set(components Quick Widgets)
elseif(CONSUMER_FRONTEND STREQUAL "Mixed")
    set(components Widgets Quick)
elseif(CONSUMER_FRONTEND STREQUAL "Widgets" OR CONSUMER_FRONTEND STREQUAL "Quick")
    set(components ${CONSUMER_FRONTEND})
else()
    message(FATAL_ERROR "Unknown consumer frontend")
endif()
set(capabilities)
foreach(frontend IN LISTS components)
    list(APPEND capabilities Frameless${frontend} Material${frontend})
endforeach()
find_package(WindowHeler 0.1 CONFIG REQUIRED COMPONENTS ${capabilities})
if(NOT TARGET WindowHeler::Runtime OR TARGET WindowHeler::FramelessQuickPlugin)
    message(FATAL_ERROR "The package must include Runtime and must not export a QML plugin")
endif()
qt_add_executable(windowheler_consumer main.cpp)
foreach(capability IN LISTS capabilities)
    target_link_libraries(windowheler_consumer PRIVATE WindowHeler::${capability})
endforeach()
if("Widgets" IN_LIST components)
    target_compile_definitions(windowheler_consumer PRIVATE CONSUMER_WIDGETS)
endif()
if("Quick" IN_LIST components)
    target_compile_definitions(windowheler_consumer PRIVATE
        CONSUMER_QUICK
        CONSUMER_QML_IMPORT_PATH="${WindowHeler_QML_IMPORT_PATH}"
    )
endif()
# The runner reads the exact executable location for any generator/configuration.
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/executable-$<CONFIG>.txt"
    CONTENT "$<TARGET_FILE:windowheler_consumer>"
)
